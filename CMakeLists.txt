cmake_minimum_required(VERSION 3.24)
project(draughts LANGUAGES CXX C)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

option(DRAUGHTS_BUILD_TESTS           "Build unit tests"          ON)
option(DRAUGHTS_BUILD_UI              "Build the Qt 6 GUI"        ON)
option(DRAUGHTS_WARNINGS_AS_ERRORS    "Treat warnings as errors"  ON)

find_package(Threads REQUIRED)

# ?????????????????????????????????????????????????????????????????????????????
# Warning flags ? applied per target below
# ?????????????????????????????????????????????????????????????????????????????
function(draughts_apply_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
        if(DRAUGHTS_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE /WX)
        endif()
    else()
        target_compile_options(${target} PRIVATE
            -Wall -Wextra -Wpedantic -Wshadow -Wconversion -Wsign-conversion)
        if(DRAUGHTS_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
    endif()
endfunction()

# ?????????????????????????????????????????????????????????????????????????????
# draughts_core ? pure rules engine (no GUI, no AI dependencies)
# ?????????????????????????????????????????????????????????????????????????????
add_library(draughts_core STATIC
    src/core/Zobrist.cpp
    src/core/Board.cpp
    src/core/CaptureRules.cpp
    src/core/MoveGenerator.cpp
    src/core/GameState.cpp
    src/core/GameEngine.cpp
    src/core/Notation.cpp
    src/core/Perft.cpp
)
target_include_directories(draughts_core PUBLIC src)
target_link_libraries(draughts_core PUBLIC Threads::Threads)
